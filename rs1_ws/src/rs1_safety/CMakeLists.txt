cmake_minimum_required(VERSION 3.10)
project(rs1_safety)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rs1_interfaces REQUIRED)
add_executable(emergency_stop src/emergency_stop.cpp)
target_compile_features(emergency_stop PRIVATE cxx_std_17)
ament_target_dependencies(emergency_stop rclcpp rs1_interfaces)
install(TARGETS emergency_stop DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_emergency_stop test/test_emergency_stop.py TIMEOUT 90
    ENV SAFETY_EXECUTABLE=$<TARGET_FILE:emergency_stop> PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
endif()
ament_package()
