# ---------------------------------------------------------------------------
# rs1_perception - ONNX Runtime inference profile  (RECOMMENDED)
#
#   pip install -r requirements/onnx.txt
#
# This is the torch-free deployment profile.  YoloNode's `onnx` backend imports
# only onnxruntime (+ numpy/opencv), so the whole inference stack is ~50 MB
# instead of the ~4.4 GB that PyTorch + torchvision + ultralytics (+ CUDA libs)
# needs.  Nothing here pulls in torch.
#
# Build the model once (on any machine) with:
#   python yolo_training/export_onnx.py --format onnx
#
# Then select it with:
#   ros2 launch rs1_perception tree_identification.launch.py backend:=onnx
# ---------------------------------------------------------------------------

onnxruntime>=1.18.0
# Humble's binary cv_bridge uses the NumPy 1.x ABI, and Jammy's SciPy (used
# by Husky's ground filter) requires NumPy <1.25. OpenCV 4.12+ requires NumPy 2.
numpy>=1.23.0,<1.25
opencv-python>=4.7.0,<4.12
